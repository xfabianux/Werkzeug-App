function loadUser(){

      user &&
      typeof user.name === "string" &&
      typeof user.group === "string" &&
      user.name.trim() &&
      user.group.trim()
    ){
      return user;
    }

    return null;

  }catch(e){

  }
}

function esc(value){

    .replace(/"/g,"&quot;")
    .replace(/'/g,"&#039;");
}

async function joinGroup(createNew){

  const name =
    document.getElementById("nameInput")
      .value
      .trim();

  let group =
    document.getElementById("groupInput")
      .value
      .trim()
      .toUpperCase();

  if(!name){

    showMessage(
      "Bitte deinen Namen eingeben.",
      "error"
    );

    return;
  }


  if(createNew){

    if(!group){

      group = createGroupCode();

    }

  }else{

    if(!group){

      showMessage(
        "Bitte den Gruppencode eingeben.",
        "error"
      );

      return;
    }


    const { data, error } = await db
      .from("members")
      .select("id")
      .eq("group_code",group)
      .limit(1);

    if(error){

      console.error(error);

      showMessage(
        "Gruppe konnte nicht geprüft werden.",
        "error"
      );

      return;
    }


    if(!data || data.length === 0){

      showMessage(
        "Diese Gruppe gibt es noch nicht. Prüfe den Code.",
        "error"
      );

      return;
    }
  }


  S = {
    name:name,
    group:group
  };

  saveUser();


  const ok =
    await ensureCurrentMember();

  if(!ok){

    clearUser();

    return;
  }


  home();
}


function createGroupCode(){

  const chars =
    "ABCDEFGHJKLMNPQRSTUVWXYZ23456789";

  let result = "";

  for(let i=0;i<6;i++){

    result +=
      chars[Math.floor(Math.random()*chars.length)];

  }

  return result;
}

function previewPhoto(event){

    event.target.files[0];

    new FileReader();


  reader.onload = function(){

      .innerHTML = `

          alt="Vorschau"
        >

      `;

  };


  reader.readAsDataURL(file);
}

   FOTO VERKLEINERN
========================================================= */

async function resizeImage(file){

  return new Promise((resolve,reject)=>{

    const reader =
      new FileReader();


    reader.onload = function(){

      const img =
        new Image();


      img.onload = function(){

        const maxSize = 700;

        let width = img.width;
        let height = img.height;


        if(width > height){

          if(width > maxSize){

            height =
              Math.round(
                height * maxSize / width
              );

            width = maxSize;
          }

        }else{

          if(height > maxSize){

            width =
              Math.round(
                width * maxSize / height
              );

            height = maxSize;
          }

        }


        const canvas =
          document.createElement("canvas");

        canvas.width = width;
        canvas.height = height;


        const ctx =
          canvas.getContext("2d");


        ctx.drawImage(
          img,
          0,
          0,
          width,
          height
        );


        resolve(
          canvas.toDataURL(
            "image/jpeg",
            0.75
          )
        );

      };


      img.onerror = reject;

      img.src = reader.result;
    };


    reader.onerror = reject;

    reader.readAsDataURL(file);

  });
}

async function saveTool(){

    document.getElementById("toolName")
      .value

      .value
      .trim();

    document.getElementById("toolCategory")

    document.getElementById("photoInput")
      .files[0];


  if(!name){

    showMessage(
      "Bitte einen Werkzeugnamen eingeben.",
      "error"
    );

    return;
  }


  if(!owner){

    showMessage(
      "Bitte Besitzer eingeben.",
      "error"
    );

    return;
  }


  const button =
    event?.target;


  if(button){

    button.disabled = true;
    button.textContent = "Speichere...";
  }


  try{

    let image_url = "";


    if(photo){

      image_url =
        await resizeImage(photo);

    }


    const { error } =
      await db
        .from("tools")
        .insert({

          group_code:S.group,

          name:name,

          category:category,

          owner:owner,

          description:description,

          image_url:image_url,

          borrower:null,

          borrowed_at:null,

          due_date:null

        });


    if(error) throw error;


    showMessage(
      "Werkzeug gespeichert ✓",
      "success"
    );


    home();


  }catch(error){

    console.error(error);

    showMessage(
      "Werkzeug konnte nicht gespeichert werden: " +
      error.message,
      "error"
    );

    if(button){

      button.disabled = false;
      button.textContent = "💾 Werkzeug speichern";

    }

  }
}

async function borrowTool(id){

  const { data, error } =
    await db
      .from("tools")
      .select("*")
      .eq("id",id)
      .eq("group_code",S.group)
      .single();


  if(error){

    showMessage(
      "Werkzeug konnte nicht gefunden werden.",
      "error"
    );

    return;
  }


  const borrower =
    prompt(
      "Wer leiht das Werkzeug aus?",
      S.name
    );


  if(!borrower || !borrower.trim()) return;


  const name =
    borrower.trim();


  const { error:updateError } =
    await db
      .from("tools")
      .update({

        borrower:name,

        borrowed_at:
          new Date().toISOString()

      })
      .eq("id",id)
      .eq("group_code",S.group);


  if(updateError){

    showMessage(
      "Ausleihen fehlgeschlagen.",
      "error"
    );

    return;
  }


  await db
    .from("loan_history")
    .insert({

      group_code:S.group,

      tool_id:id,

      tool_name:data.name,

      borrower:name,

      borrowed_at:
        new Date().toISOString(),

      due_date:null

    });


  showMessage(
    "Werkzeug wurde ausgeliehen ✓",
    "success"
  );


  home();
}
